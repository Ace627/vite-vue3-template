# 目录结构说明

> 本文件梳理项目整体目录与每个文件的作用。注释以「简单明了」为目标，便于快速定位与协作。
> 约定：`node_modules/`（依赖）、`dist/`（构建产物）为自动生成，不纳入说明。

## 一、根目录

```
vite-vue3-template/
├── AGENTS.md                  # AI 代码约束：图标统一 SvgIcon、复用 @/utils、禁止接后端等协作红线
├── README.md                  # 项目说明文档（启动方式、特性概览）
├── index.html                 # HTML 入口：挂载 #app、引入 reset/app-loading 样式、favicon
├── package.json               # 依赖清单与脚本（dev/build/preview/clean:svg/format 等）
├── pnpm-lock.yaml             # 锁定依赖的精确版本，保证可复现安装
├── pnpm-workspace.yaml        # pnpm 工作区配置（关闭 @parcel/watcher 构建）
├── oxfmt.config.ts            # 代码格式化配置（单引号/无分号/行宽 160/忽略自动生成文件）
├── vite.config.ts             # Vite 构建配置：别名、分包、gzip、反向代理、SCSS 变量注入
├── uno.config.ts              # UnoCSS 原子化 CSS 配置：预设、自定义 rules、shortcuts
├── tsconfig.json              # TS 总入口：引用 app/node 两个子配置（项目引用）
├── tsconfig.app.json          # 浏览器端 TS 配置：编译 src，含 @/ 别名、严格检查
├── tsconfig.node.json         # Node 端 TS 配置：编译 vite.config / build / oxfmt（nodenext）
├── .env                       # 公共环境变量（标题、输出目录、部署基路径、路由模式等）
├── .env.development           # 开发环境变量（端口、后端代理地址、dev-api 路径）
├── .env.production            # 生产环境变量（移除 console/debugger、生产 api 路径）
├── .gitignore                 # git 忽略规则（依赖、产物、缓存等）
├── .gitattributes             # git 属性配置（换行符等）
├── build/                     # Vite 插件抽离（见下方 build 章节）
├── docs/                      # 工程化文档集合（见下方 docs 章节）
├── public/                    # 静态资源：直接拷贝到产物根，不经打包
├── scripts/                   # 工程辅助脚本
└── src/                       # 源码主目录（见下方 src 章节）
```

## 二、build/ — Vite 插件抽离

```
build/
├── plugins/
│   ├── index.ts              # 汇总注册全部 Vite 插件（setupVitePlugins），vite.config 调用入口
│   ├── auto-import-plugin.ts # 配置 unplugin-auto-import / unplugin-vue-components 按需自动导入
│   ├── dist-compression.ts   # 构建期 gzip 预压缩（>10KB 的文本资源，level 9）
│   └── svg-icons-plugin.ts   # 注册 vite-plugin-svg-icons，把 svg-icons 目录编译成雪碧图
```

## 三、docs/ — 工程化文档

```
docs/
├── 前端工程化.md              # 总纲：项目约定、技术栈、工程化落地实践汇总
├── UI组件/                    # 组件相关文档：ElementPlus 按需/主题、SvgIcon、全局组件、ProTable、消息反馈
├── 响应式/                    # 设备自适应（移动端断点、布局切换）文档
├── 构建与工程化/              # Vite 插件设计、路径别名、gzip、分包、移除调试语句、设置中心等
└── 用户体验/                 # 主题切换、动态标题、系统水印、首屏加载动画文档
```

## 四、public/ — 静态资源（不经打包，原样拷贝）

```
public/
├── css/
│   ├── reset.css             # 浏览器样式重置，index.html 用 <link> 引入（防 FOUC，oxfmt 已忽略）
│   └── app-loading.css       # 首屏白屏阶段的加载动画样式（#app .app-loading）
└── favicon.svg               # 网站图标（被 index.html、登录页、404 页、AppLogo 引用）
```

## 五、scripts/ — 工程辅助脚本

```
scripts/
└── svg-clean.ts              # 清理新增 SVG 的冗余属性（fill/class/width 等），适配 fill:currentColor
                              # 执行：pnpm clean:svg（新增图标后必跑，保证主题色随 currentColor 变化）
```

## 六、src/ — 源码主目录

```
src/
├── main.ts                   # 应用入口：异步 bootstrap 依次装配 插件→状态→路由→挂载
├── App.vue                   # 根组件：el-config-provider（中文+全局尺寸）包裹 router-view
├── AppLogo/types.ts          # AppLogo 组件 props 类型（showTitle / title）
├── defaultSettings.ts        # 系统设置默认值与 SystemSetting 类型（主题/尺寸/水印/转场等）
├── assets/                   # 静态素材
│   ├── images/
│   │   └── 404.png           # 404 页面用到的占位图片
│   └── svg-icons/            # 图标库（共 74 个 .svg）
│                              # 命名约定：文件名即图标名，<SvgIcon name="Home" /> 对应 Home.svg
│                              # 全部经 pnpm clean:svg 清理，fill:currentColor 随主题变色
├── components/               # 全局/通用组件（均经 global-component.ts 显式注册为全局组件）
│   ├── SvgIcon/
│   │   ├── index.vue         # 图标渲染组件：<use href="#icon-name">，支持 color/size
│   │   └── types.ts          # SvgIcon 组件 props 类型（name/color/size）
│   ├── ProPagination/
│   │   └── index.vue         # 分页组件：基于 el-pagination 封装，双向绑定页码/页大小 + 移动端适配
│   └── ProTable/
│       ├── index.vue         # 表格组件：基于 el-table 封装，动态列 + 插槽透传 + 实例代理
│       └── types.ts          # ProTable 列配置与 props 类型（泛型约束 prop）
├── hooks/                    # 组合式函数（自动导入，无需手动 import）
│   ├── useTheme.ts           # 主题切换：切 .dark 类 + View Transitions 圆形扩散 + 防 FOUC
│   ├── useDynamicTitle.ts    # 动态标题：按路由 meta.title 拼接 document.title
│   └── useResize.ts          # 响应式：监听窗口宽度设置桌面/移动端布局
├── layout/                   # 整体布局框架
│   ├── index.vue             # 布局根：el-watermark 包裹全站 + Navbar + AppMain
│   └── components/
│       ├── AppLogo/          # 顶部品牌区（logo + 标题）
│       │   ├── index.vue     # AppLogo 视图
│       │   └── types.ts      # AppLogo props 类型
│       ├── AppMain/          # 主内容区：router-view + 路由转场 Transition
│       │   └── index.vue
│       └── Navbar/           # 顶部导航栏：品牌 + 设置入口 + 主题切换
│           ├── index.vue
│           └── ThemeSwitch.vue  # 明暗切换按钮，委托 useTheme().toggleTheme
├── plugins/                  # 应用插件装配
│   ├── index.ts              # setupPlugins：批量注册全局组件
│   └── modules/
│       └── global-component.ts  # 注册 SvgIcon / ProPagination / ProTable 为全局组件
├── router/                   # 路由
│   ├── index.ts              # 创建 router 实例 + setupRouter（挂守卫、isReady 后挂载）
│   ├── router.guard.ts       # 全局前置守卫：白名单 /login、无 token 跳登录并携带 redirect
│   └── modules/
│       └── static.route.ts   # 静态路由表（dashboard/settings/login/404，无动态权限）
├── store/                    # 状态管理（Pinia）
│   ├── index.ts              # 创建 Pinia 实例 + setupStore 挂载
│   └── modules/
│       ├── app.ts            # app store：设备类型、侧栏折叠状态
│       └── setting.ts        # setting store：系统设置读写、保存/重置（落 localStorage）
├── styles/                   # 全局样式
│   ├── index.scss            # 样式入口：引入变量/转场/暗黑主题 + body 背景 + .app-content 容器
│   ├── variables.scss        # SCSS 变量（导航栏高度/背景/阴影等）
│   ├── transition.scss       # 路由转场动画（fade-transform / sidebar-logo-fade）
│   └── element-plus/
│       ├── el-theme-light.scss  # Element Plus 亮色主题变量（primary #0077ff 等）
│       └── el-theme-dark.scss   # Element Plus 暗色主题（引入官方 dark css-vars）
├── types/                    # TypeScript 类型声明
│   ├── api/
│   │   ├── api.d.ts          # 通用接口响应结构 ApiResponse<T>
│   │   └── auth.ts           # 登录参数类型 LoginParams
│   ├── auto-generate/        # 自动生成的类型（勿手改）
│   │   ├── auto-import.d.ts  # unplugin-auto-import 自动导入的全局 API 声明
│   │   └── auto-components.d.ts  # 自动识别的全局组件声明（ElButton/ElTooltip 等）
│   ├── global/
│   │   ├── vite-env.d.ts     # 扩展 import.meta.env 自定义环境变量类型
│   │   ├── module.d.ts       # 声明虚拟模块 virtual:svg-icons-register / -names
│   │   ├── vue-router.d.ts   # 扩展 RouteMeta（title/icon/hidden）
│   │   └── global-component.d.ts  # 声明全局组件 SvgIcon / ProPagination / ProTable 类型
│   └── utils/                # 工具相关类型（当前为空占位）
├── utils/                    # 工具方法（统一从 @/utils 引入）
│   ├── index.ts              # barrel 入口：re-export 各工具 + 暴露 isString
│   ├── tip-modal.ts          # 统一消息反馈（ElMessage/MessageBox/Notification/Loading 封装）
│   ├── validate.ts           # 校验工具（isString 等）
│   ├── storage-cache.util.ts # localStorage 封装 StorageCache（带前缀/过期/JSON 序列化）
│   ├── cache/
│   │   ├── token.cache.ts    # token 持久化（access/refresh 读写删，基于 StorageCache）
│   │   └── system-setting.cache.ts  # 系统设置持久化（基于 StorageCache）
│   └── request/              # axios 请求层
│       ├── index.ts          # 创建 axios 实例并注册三个拦截器
│       └── interceptor/
│           ├── jwt-auth.ts           # 请求拦截：自动携带 Authorization Bearer token
│           ├── response-transform.ts # 响应拦截：剥离 code/message、取 data、处理二进制 JSON
│           └── response-error.ts     # 响应错误拦截：统一提示 + 401 清 token 并重载
└── views/                    # 页面视图（路由挂载的组件）
    ├── core/
    │   ├── login.vue         # 登录页（mock：写死 token 回跳 redirect）
    │   └── 404.vue           # 404 页面（含移动端适配）
    ├── dashboard/
    │   └── index.vue         # 首页（演示用，含退出登录）
    └── settings/
        └── index.vue         # 设置页：主题/尺寸/动态标题/水印/转场 + 保存/重置
```
